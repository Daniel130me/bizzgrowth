   375→const SYNC_SOURCES = [
   376→  { name: "Shopify", detail: "Orders & checkout analytics — importing the last 3 months" },
   377→  { name: "Paystack", detail: "Payments — success rates, failure reasons and settlement totals" },
   378→];
   379→
   380→function SyncingScreen() {
   381→  const navigate = useBizz((s) => s.navigate);
   382→  const [progress, setProgress] = useState(8);
   383→  const [done, setDone] = useState(false);
   384→  const mountedRef = useRef(true);
   385→
   386→  useEffect(() => {
   387→    mountedRef.current = true;
   388→    const iv = window.setInterval(() => {
   389→      setProgress((p) => (p >= 100 ? 100 : Math.min(100, p + 4 + Math.round(Math.random() * 7))));
   390→    }, 550);
   391→    const t = window.setTimeout(() => {
   392→      if (!mountedRef.current) return;
   393→      setProgress(100);
   394→      setDone(true);
   395→      toast.success("Sync complete", {
   396→        description: "Orders, payments and workspace metrics are up to date.",
   397→      });
   398→    }, 10000);
   399→    return () => {
   400→      mountedRef.current = false;
   401→      window.clearInterval(iv);
   402→      window.clearTimeout(t);
   403→    };
   404→  }, []);
   405→
   406→  const shopifyDone = progress >= 60;
   407→  const paystackDone = done;
   408→
   409→  return (
   410→    <div className="mx-auto w-full max-w-3xl">
   411→      <PageHeader
   412→        title="Importing your data"
   413→        description="This usually takes under a minute. You can leave this screen — we'll notify you when it's ready."
   414→        badge={<SystemStateChip />}
   415→      />
   416→
   417→      <SectionCard title="Sync progress" description="Workspace-level import across connected sources.">
   418→        <div className="flex items-center gap-3">
   419→          {done ? (
   420→            <span className="flex size-10 items-center justify-center rounded-full bg-emerald-50 text-emerald-600">
   421→              <CheckCircle2 className="size-5" aria-hidden />
   422→            </span>
   423→          ) : (
   424→            <Loader2 className="size-6 animate-spin text-blue-600" aria-hidden />
   425→          )}
   426→          <div className="flex-1">
   427→            <div className="flex items-baseline justify-between">
   428→              <p className="text-sm font-medium">{done ? "Sync complete" : "Importing your data…"}</p>
   429→              <span className="text-sm font-medium text-muted-foreground tnum">{progress}%</span>
   430→            </div>
   431→            <Progress value={progress} className="mt-2 h-2" aria-label={`Sync progress ${progress}%`} />
   432→          </div>
   433→        </div>
   434→
   435→        <Separator className="my-4" />
   436→
   437→        <ul className="space-y-3">
   438→          {SYNC_SOURCES.map((src, i) => {
   439→            const isDone = i === 0 ? shopifyDone : paystackDone;
   440→            return (
   441→              <li key={src.name} className="flex items-start gap-3">
   442→                <span className="mt-0.5">
   443→                  {isDone ? (
   444→                    <CheckCircle2 className="size-4.5 text-emerald-600" aria-label={`${src.name} complete`} />
   445→                  ) : (
   446→                    <Loader2 className="size-4.5 animate-spin text-blue-600" aria-label={`${src.name} syncing`} />
   447→                  )}
   448→                </span>
   449→                <div>
   450→                  <p className="text-sm font-medium">
   451→                    {src.name}
   452→                    <span className="ml-2 text-xs font-normal text-muted-foreground">{isDone ? "Done" : "Syncing…"}</span>
   453→                  </p>
   454→                  <p className="text-xs text-muted-foreground">{src.detail}</p>
   455→                </div>
   456→              </li>
   457→            );
   458→          })}
   459→        </ul>
   460→      </SectionCard>
   461→
   462→      {done ? (
   463→        <InfoBanner tone="success" title="Your data is up to date" >
   464→          Metrics, funnel charts and diagnoses now include everything just imported.
   465→        </InfoBanner>
   466→      ) : (
   467→        <InfoBanner tone="info" title="You can leave this screen">
   468→          The sync continues in the background and we&apos;ll notify you the moment it&apos;s ready — nothing is lost by navigating away.
   469→        </InfoBanner>
   470→      )}
   471→
   472→      <div className="mt-5 flex flex-col gap-2 sm:flex-row">
   473→        <Button variant="outline" className="flex-1" onClick={() => navigate("home")}>
   474→          Continue in background
   475→        </Button>
   476→        <Button className="flex-1" onClick={() => navigate("data")} variant={done ? "default" : "ghost"}>
   477→          {done ? "View performance" : "Go to Data Centre"}
   478→        </Button>
   479→      </div>
   480→    </div>
   481→  );
   482→}
