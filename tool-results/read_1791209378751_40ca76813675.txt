   217→interface Matcher {
   218→  test: RegExp;
   219→  content: AnswerContent;
   220→}
   221→
   222→const MATCHERS: Record<BusinessType, { list: Matcher[]; fallback: AnswerContent }> = {
   223→  service: {
   224→    list: [
   225→      { test: /enquir|customer|sales|convers|lead/i, content: SH_ENQUIRY },
   226→      { test: /channel|best|perform|platform/i, content: SH_CHANNEL },
   227→      { test: /tried|before|experiment|test|histor/i, content: SH_TRIED },
   228→      { test: /chang|month|trend|happen/i, content: SH_CHANGED },
   229→      { test: /focus|week|priorit|next/i, content: SH_FOCUS },
   230→    ],
   231→    fallback: SH_FOCUS,
   232→  },
   233→  ecommerce: {
   234→    list: [
   235→      { test: /order|checkout|payment|cart|sales|customer|convers|traffic|enquir/i, content: KH_ORDERS },
   236→      { test: /channel|best|perform|ads|platform/i, content: KH_CHANNEL },
   237→      { test: /tried|before|experiment|test|histor/i, content: KH_TRIED },
   238→      { test: /chang|month|trend|happen/i, content: KH_CHANGED },
   239→      { test: /focus|week|priorit|next/i, content: KH_FOCUS },
   240→    ],
   241→    fallback: KH_FOCUS,
   242→  },
   243→};
   244→
   245→function answerFor(question: string, type: BusinessType): AnswerContent {
   246→  const m = MATCHERS[type];
   247→  const hit = m.list.find((c) => c.test.test(question));
   248→  return hit ? hit.content : m.fallback;
   249→}

   264→function AskScreen() {
   265→  const businessId = useBizz((s) => s.businessId);
   266→  const userWorkspace = useBizz((s) => s.userWorkspace);
   267→  const navigate = useBizz((s) => s.navigate);
   268→  const business = getBusiness(businessId, userWorkspace);
   269→
   270→  const [input, setInput] = useState("");
   271→  const [typing, setTyping] = useState(false);
   272→  const [thread, setThread] = useState<ThreadItem[]>([]);
   273→  const inputRef = useRef<HTMLInputElement>(null);
   274→  const timerRef = useRef<ReturnType<typeof setTimeout> | null>(null);
   275→
   276→  useEffect(() => {
   277→    return () => {
   278→      if (timerRef.current) clearTimeout(timerRef.current);
   279→    };
   280→  }, []);
   281→
   282→  const suggestions =
   283→    business.type === "service"
   284→      ? [
   285→          "Why are we getting more enquiries but not more customers?",
   286→          "What should I focus on this week?",
   287→          "Which channel is performing best?",
   288→          "What changed this month?",
   289→          "What have we tried before?",
   290→        ]
   291→      : [
   292→          "Why is traffic growing but orders falling?",
   293→          "What should I focus on this week?",
   294→          "Which channel is performing best?",
   295→          "What changed this month?",
   296→          "What have we tried before?",
   297→        ];
   298→
   299→  function ask(raw: string) {
   300→    const q = raw.trim();
   301→    if (!q) {
   302→      toast.error("Type a question first", { description: "Ask in your own words, or pick a suggestion." });
   303→      return;
   304→    }
   305→    if (typing) return;
   306→    setInput("");
   307→    setThread((t) => [...t, { kind: "question", text: q }]);
   308→    setTyping(true);
   309→    timerRef.current = setTimeout(() => {
   310→      setTyping(false);
   311→      setThread((t) => [...t, { kind: "answer", question: q, content: answerFor(q, business.type) }]);
   312→    }, 1000);
   313→  }
   314→
   315→  return (
   316→    <div className="mx-auto w-full max-w-4xl">
   317→      {/* Hero — expressive but serious */}
   318→      <Card className="overflow-hidden border-violet-200 bg-gradient-to-br from-violet-50 via-white to-white p-0">

   328→                  Grounded answers from {business.name}&apos;s workspace data.
   329→                </p>
   330→              </div>
   331→            </div>
   332→            <Button variant="outline" size="sm" onClick={() => navigate("ask/history")}>
   333→              <History className="mr-1.5 size-4" aria-hidden /> Conversation history
   334→            </Button>
   335→          </div>
   336→          <form
   337→            className="mt-4 flex flex-col gap-2 sm:flex-row"
   338→            onSubmit={(e) => {
   339→              e.preventDefault();
   340→              ask(input);
   341→            }}
   342→          >
   343→            <label htmlFor="ask-input" className="sr-only">
   344→              Ask a question about your business
   345→            </label>
   346→            <Input
   347→              id="ask-input"
   348→              ref={inputRef}
   349→              value={input}
   350→              onChange={(e) => setInput(e.target.value)}
   351→              placeholder="e.g. Why are we getting more enquiries but not more customers?"
   352→              className="h-12 flex-1 border-violet-200 bg-white text-base"
   353→              autoComplete="off"
   354→            />
   355→            <Button type="submit" className="h-12 bg-violet-600 px-6 text-base hover:bg-violet-700" disabled={typing}>
   356→              <Send className="mr-2 size-4" aria-hidden /> Ask
   357→            </Button>
   358→          </form>
   359→          <div className="mt-3 flex flex-wrap gap-2">
   360→            {suggestions.map((s) => (
   361→              <button
   362→                key={s}
   363→                onClick={() => ask(s)}
   364→                disabled={typing}
   365→                className="rounded-full border border-violet-200 bg-white px-3 py-1.5 text-xs font-medium text-violet-700 transition-colors hover:bg-violet-100 disabled:opacity-50"
   366→              >
   367→                {s}
   368→              </button>
   369→            ))}
   370→          </div>
   371→        </div>
   372→        <p className="px-6 py-3 text-xs text-muted-foreground">
   373→          Answers are grounded in your workspace data and marked with sources. AI interpretation is labelled as such.
   374→        </p>
   375→      </Card>
   376→
   377→      {/* Thread */}
   378→      <div className="mt-6 space-y-4">
   379→        {thread.length === 0 && !typing ? (
   380→          <EmptyState
   381→            icon={<MessageCircleQuestion className="size-5" />}
   382→            title="Start with a question"
   383→            description="Try a suggestion above, or ask in your own words. Every answer references the actual numbers recorded in this workspace and shows where each fact came from."
   384→          />
   385→        ) : null}
   386→
   387→        {thread.map((item, i) =>
   388→          item.kind === "question" ? (
   389→            <div key={i} className="flex justify-end">
   390→              <p className="max-w-[85%] rounded-2xl rounded-br-md bg-blue-600 px-4 py-2.5 text-sm leading-relaxed text-white sm:max-w-[70%]">
   391→                {item.text}
   392→              </p>
   393→            </div>
   394→          ) : (
   395→            <AnswerCard key={i} answer={item.content} question={item.question} onFollowUp={() => inputRef.current?.focus()} />
   396→          )
   397→        )}
   398→
   399→        {typing ? (
   400→          <Card className="flex items-center gap-3 p-4" role="status" aria-label="Preparing answer">
   401→            <div className="flex size-8 items-center justify-center rounded-lg bg-violet-600 text-white">
   402→              <Sparkles className="size-4" aria-hidden />
   403→            </div>
   404→            <div className="flex items-center gap-1.5" aria-hidden>
   405→              <span className="size-2 animate-bounce rounded-full bg-violet-400 [animation-delay:0ms]" />
   406→              <span className="size-2 animate-bounce rounded-full bg-violet-400 [animation-delay:150ms]" />
   407→              <span className="size-2 animate-bounce rounded-full bg-violet-400 [animation-delay:300ms]" />
   408→            </div>
   409→            <span className="text-sm text-muted-foreground">Reviewing your workspace data…</span>
   410→          </Card>
   411→        ) : null}
   412→      </div>
   413→    </div>
   414→  );
   415→}
