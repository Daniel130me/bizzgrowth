    58→function CapabilityBadge({ badge }: { badge: IntegrationDef["badge"] }) {
    59→  const cls =
    60→    badge === "Read"
    61→      ? "bg-blue-50 text-blue-700 border-blue-200"
    62→      : badge === "Observe"
    63→        ? "bg-slate-100 text-slate-600 border-slate-200"
    64→        : "bg-amber-50 text-amber-700 border-amber-200";
    65→  const icon = badge === "Read" ? <Unlock className="size-3" aria-hidden /> : badge === "Observe" ? <Eye className="size-3" aria-hidden /> : <PenLine className="size-3" aria-hidden />;
    66→  return (
    67→    <Badge variant="outline" className={cn("gap-1 font-normal", cls)}>
    68→      {icon}
    69→      {badge}
    70→    </Badge>
    71→  );
    72→}

   427→const SYNC_STEPS = ["Authorizing", "Connecting", "Importing data", "Calculating metrics", "Checking history", "Updating diagnosis"];
   428→
   429→function SyncingScreen() {
   430→  const navigate = useBizz((s) => s.navigate);
   431→  const setConnection = useBizz((s) => s.setConnection);
   432→  const [step, setStep] = useState(1); // step 0 (Authorizing) is already complete
   433→  const finishedRef = useRef(false);
   434→  const pendingId = getPendingConnection();
   435→  const int = integrationById(pendingId);
   436→  const name = int?.name ?? "Source";
   437→
   438→  useEffect(() => {
   439→    if (step >= SYNC_STEPS.length) {
   440→      if (finishedRef.current) return;
   441→      const t = setTimeout(() => {
   442→        if (finishedRef.current) return;
   443→        finishedRef.current = true;
   444→        setConnection(pendingId, "connected");
   445→        toast.success(`${name} connected — first insights arriving`, {
   446→          description: "Initial history is imported; metrics update on the next pass.",
   447→        });
   448→        navigate(`source-settings/${pendingId}`);
   449→      }, 900);
   450→      return () => clearTimeout(t);
   451→    }
   452→    const t = setTimeout(() => setStep((s) => s + 1), 2000);
   453→    return () => clearTimeout(t);
   454→  }, [step, pendingId, name, setConnection, navigate]);
   455→
   456→  const progress = Math.round(((step + 0.5) / SYNC_STEPS.length) * 100);
   457→
   458→  return (
   459→    <div className="mx-auto w-full max-w-2xl">
   460→      <Card className="p-6">
   461→        <div className="flex items-start gap-3 rounded-xl border border-emerald-200 bg-emerald-50/70 p-4">
   462→          <CheckCircle2 className="mt-0.5 size-5 shrink-0 text-emerald-600" aria-hidden />
   463→          <div>
   464→            <h1 className="text-base font-semibold text-emerald-900">{name} authorized successfully</h1>
   465→            <p className="mt-0.5 text-sm text-emerald-800/80">
   466→              Read-only connection established. First sync is running — this usually takes under a minute.
   467→            </p>
   468→          </div>
   469→        </div>
   470→
   471→        <div className="mt-6">
   472→          <div className="flex items-center justify-between text-sm">
   473→            <span className="font-medium">Initial sync</span>
   474→            <span className="tnum text-muted-foreground">{progress}%</span>
   475→          </div>
   476→          <Progress value={progress} className="mt-2 h-2 [&>div]:bg-blue-600" aria-label={`Sync progress ${progress}%`} />
   477→        </div>
   478→
   479→        <ol className="mt-6 space-y-3" aria-label="Sync steps">
   480→          {SYNC_STEPS.map((label, i) => {
   481→            const done = i < step;
   482→            const active = i === step;
   483→            return (
   484→              <li key={label} className="flex items-center gap-3">
   485→                <span
   486→                  className={cn(
   487→                    "flex size-7 shrink-0 items-center justify-center rounded-full border",
   488→                    done && "border-emerald-200 bg-emerald-50 text-emerald-600",
   489→                    active && "border-blue-200 bg-blue-50 text-blue-600",
   490→                    !done && !active && "border-slate-200 bg-slate-50 text-slate-400"
   491→                  )}
   492→                  aria-hidden
   493→                >
   494→                  {done ? (
   495→                    <Check className="size-4" />
   496→                  ) : active ? (
   497→                    <RefreshCw className="size-4 animate-spin" />
   498→                  ) : (
   499→                    <span className="text-xs tnum">{i + 1}</span>
   500→                  )}
   501→                </span>
   502→                <span className={cn("text-sm", done && "text-muted-foreground", active && "font-medium", !done && !active && "text-slate-400")}>
   503→                  {label}
   504→                  {done && i === 0 ? " — token verified" : ""}
   505→                </span>
   506→                {active ? (
   507→                  <span className="ml-auto text-xs text-muted-foreground" role="status">
   508→                    in progress…
   509→                  </span>
   510→                ) : null}
   511→              </li>
   512→            );
   513→          })}
   514→        </ol>

   608→function SourceSettingsScreen({ params }: { params: Record<string, string> }) {
   609→  const businessId = useBizz((s) => s.businessId);
   610→  const userWorkspace = useBizz((s) => s.userWorkspace);
   611→  const connections = useBizz((s) => s.connections);
   612→  const setConnection = useBizz((s) => s.setConnection);
   613→  const navigate = useBizz((s) => s.navigate);
   614→  const business = getBusiness(businessId, userWorkspace);
   615→  const int = integrationById(params.id);
   616→
   617→  const [syncingNow, setSyncingNow] = useState(false);
   618→  const [lastSyncOverride, setLastSyncOverride] = useState<string | null>(null);
   619→  const syncTimer = useRef<ReturnType<typeof setTimeout> | null>(null);
   620→
   621→  useEffect(() => {
   622→    return () => {
   623→      if (syncTimer.current) clearTimeout(syncTimer.current);
   624→    };
   625→  }, []);
   626→
   627→  if (!int) {
   628→    return (
   629→      <div className="mx-auto w-full max-w-3xl">
   630→        <PageHeader title="Connected source" />
   631→        <EmptyState
   632→          icon={<Plug className="size-5" />}
   633→          title="Source not found"
   634→          description="We couldn't find that source. Pick a source from the library instead."
   635→          actions={
   636→            <Button size="sm" onClick={() => navigate("integrations")}>
   637→              Open sources library
   638→            </Button>
   639→          }
   640→        />
   641→      </div>
   642→    );
   643→  }
   644→
   645→  const status = effectiveStatus(int, connections);
   646→
   647→  if (status === "available") {
   648→    return (
   649→      <div className="mx-auto w-full max-w-3xl">
   650→        <PageHeader title={`${int.name} settings`} back={<Button variant="ghost" size="sm" className="-ml-2 mb-2" onClick={() => navigate("integrations")}>Back to sources</Button>} />
   651→        <EmptyState
   652→          icon={<Plug className="size-5" />}
   653→          title={`${int.name} is not connected yet`}
   654→          description="Connect this source to see settings, sync history and health."
   655→          actions={
   656→            <Button size="sm" onClick={() => navigate(`connect/${int.id}`)}>
   657→              <Plug className="mr-1.5 size-4" aria-hidden /> Connect {int.name}
   658→            </Button>
   659→          }
   660→        />
   661→      </div>
   662→    );
   663→  }
   664→
   665→  function syncNow() {
   666→    setSyncingNow(true);
   667→    syncTimer.current = setTimeout(() => {
   668→      setSyncingNow(false);
   669→      setLastSyncOverride("Just now");
   670→      toast.success("Sync complete", { description: `${int?.name ?? "Source"} data is up to date.` });
   671→    }, 2000);
   672→  }
   673→
   674→  function reconnect() {
   675→    setPendingConnection(int!.id);
   676→    setConnection(int!.id, "syncing");
   677→    navigate("syncing");
   678→  }
   679→
   680→  function disconnect() {
   681→    setConnection(int!.id, "available");
   682→    toast("Source disconnected", { description: `${int?.name ?? "Source"} data is retained for 30 days.` });
   683→    navigate("integrations");
   684→  }
   685→
   686→  return (
   687→    <div className="mx-auto w-full max-w-3xl">
   688→      <PageHeader
   689→        title={`${int.name} settings`}
   690→        description={int.description}
   691→        back={
   692→          <Button variant="ghost" size="sm" className="-ml-2 mb-2" onClick={() => navigate("integrations")}>
   693→            Back to sources
   694→          </Button>
   695→        }
   696→        badge={<StatusBadge status={status} />}
   697→      />
   698→
   699→      <div className="space-y-5">
   700→        <SectionCard title="Account">
   701→          <div className="space-y-2">
   702→            <KeyValue label="Account name">{accountNameFor(int, business)}</KeyValue>
   703→            <KeyValue label="Connected since">{int.connectedSince ? fmtDate(int.connectedSince) : "—"}</KeyValue>
   704→            <KeyValue label="Last sync">
   705→              {lastSyncOverride ?? (status === "error" ? int.lastSync ?? "—" : int.lastSync ?? "A few minutes ago")}
   706→            </KeyValue>
   707→            <KeyValue label="Next sync">{status === "error" ? "paused" : int.nextSync ?? "In about an hour"}</KeyValue>
   708→          </div>
   709→        </SectionCard>
   710→
   711→        <SectionCard title="Health">
   712→          <div className="flex flex-wrap items-center gap-3">
   713→            <StatusBadge status={status} />
   714→            <p className="text-sm text-muted-foreground">
   715→              {status === "connected"
   716→                ? "Syncing normally. Data lands in your workspace within minutes of each sync window."
   717→                : status === "error"
   718→                  ? "Last sync failed — authorization needs to be refreshed. Historical data is retained."
   719→                  : "Initial sync in progress."}
   720→            </p>
   721→          </div>
   722→        </SectionCard>
   723→
   724→        <SectionCard title="Permissions" description="Read-only — these cannot be widened from the prototype.">
   725→          <div className="grid gap-4 sm:grid-cols-2">
   726→            <div>
   727→              <p className="text-xs font-semibold uppercase tracking-wide text-blue-700">Reads</p>
   728→              <ul className="mt-1.5 space-y-1">
   729→                {int.permissions.read.map((r) => (
   730→                  <li key={r} className="text-sm text-muted-foreground">
   731→                    · {r}
   732→                  </li>
   733→                ))}
   734→              </ul>
   735→            </div>
   736→            <div>
   737→              <p className="text-xs font-semibold uppercase tracking-wide text-slate-600">Cannot</p>
   738→              <ul className="mt-1.5 space-y-1">
   739→                {int.permissions.cannot.map((c) => (
   740→                  <li key={c} className="text-sm text-muted-foreground">
   741→                    · {c}
   742→                  </li>
   743→                ))}
   744→              </ul>
   745→            </div>
   746→          </div>
   747→        </SectionCard>
   748→
   749→        <SectionCard title="Data types">
   750→          <div className="flex flex-wrap gap-2">
   751→            {int.provides.map((p) => (

   756→          </div>
   757→        </SectionCard>
   758→      </div>
   759→
   760→      <div className="mt-6 flex flex-col gap-2 sm:flex-row">
   761→        <Button onClick={syncNow} disabled={syncingNow} className="flex-1">
   762→          {syncingNow ? (
   763→            <>
   764→              <RefreshCw className="mr-1.5 size-4 animate-spin" aria-hidden /> Syncing…
   765→            </>
   766→          ) : (
   767→            <>
   768→              <RefreshCw className="mr-1.5 size-4" aria-hidden /> Sync now
   769→            </>
   770→          )}
   771→        </Button>
   772→        <Button variant="outline" className="flex-1" onClick={reconnect}>
   773→          Reconnect
   774→        </Button>
   775→        <AlertDialog>
   776→          <AlertDialogTrigger asChild>
   777→            <Button variant="destructive" className="flex-1">
   778→              Disconnect
   779→            </Button>
   780→          </AlertDialogTrigger>
   781→          <AlertDialogContent>
   782→            <AlertDialogHeader>
   783→              <AlertDialogTitle>Disconnect {int.name}?</AlertDialogTitle>
   784→              <AlertDialogDescription>
   785→                {accountNameFor(int, business)} will stop syncing. Analyses that use this source will note the gap until you
   786→                reconnect. Imported data is retained for 30 days.
   787→              </AlertDialogDescription>
   788→            </AlertDialogHeader>
   789→            <AlertDialogFooter>
   790→              <AlertDialogCancel>Cancel</AlertDialogCancel>
   791→              <AlertDialogAction onClick={disconnect} className="bg-red-600 hover:bg-red-700">
   792→                Disconnect
   793→              </AlertDialogAction>
   794→            </AlertDialogFooter>
   795→          </AlertDialogContent>
   796→        </AlertDialog>
   797→      </div>
   798→
   799→      {status === "error" ? (
   800→        <button
   801→          onClick={() => navigate("connection-error")}
   802→          className="mt-4 flex w-full items-center justify-between rounded-xl border border-red-200 bg-red-50/60 p-4 text-left transition-colors hover:bg-red-50"
   803→        >
   804→          <span className="flex items-center gap-2 text-sm font-medium text-red-800">
   805→            <CircleAlert className="size-4" aria-hidden /> Authorization expired — reconnect required
   806→          </span>
   807→          <ChevronRight className="size-4 text-red-400" aria-hidden />

   809→      ) : null}
   810→
   811→      <button
   812→        onClick={() => navigate(`integrations/${int.id}`)}
   813→        className="mt-4 inline-flex items-center gap-1 text-sm font-medium text-blue-700 hover:underline"

   815→        <ArrowLeft className="size-4" aria-hidden /> What {int.name} provides and unlocks
