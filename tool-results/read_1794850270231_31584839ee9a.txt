   248→function answerFor(question: string, type: BusinessType): AnswerContent {
   249→  const m = MATCHERS[type];
   250→  const hit = m.list.find((c) => c.test.test(question));
   251→  return hit ? hit.content : m.fallback;
   252→}

   458→  function ask(raw: string) {
   459→    const q = raw.trim();
   460→    if (!q) {
   461→      toast.error("Type a question first", { description: "Ask in your own words, or pick a suggestion." });
   462→      return;
   463→    }
   464→    if (typing) return;
   465→    setInput("");
   466→    setThread((t) => {
   467→      const next: ThreadItem[] = [...t, { kind: "question" as const, text: q }];
   468→      threadRef.current = next;
   469→      return next;
   470→    });
   471→    setTyping(true);
   472→    void (async () => {
   473→      try {
   474→        const res = await fetch("/api/ask", {
   475→          method: "POST",
   476→          headers: { "Content-Type": "application/json" },
   477→          body: JSON.stringify({
   478→            question: q,
   479→            history: threadRef.current
   480→              .filter((item): item is ThreadAnswer => item.kind === "answer")
   481→              .slice(-3)
   482→              .map((item) => ({ question: item.question, summary: item.content.grounded.join(" ").slice(0, 200) })),
   483→            workspace: buildContext(),
   484→          }),
   485→        });
   486→        const data = (await res.json()) as { ok?: boolean; answer?: AnswerContent };
   487→        if (data?.ok && data.answer && Array.isArray(data.answer.grounded) && data.answer.grounded.length > 0) {
   488→          pushAnswer(q, data.answer);
   489→          return;
   490→        }
   491→        throw new Error("no answer");
   492→      } catch {
   493→        // Offline fallback — the owner's own facts for user workspaces,
   494→        // canned demo-dataset answers for the sample businesses.
   495→        pushAnswer(q, businessId === "user-workspace" ? userFallbackAnswer(q) : answerFor(q, business.type));
   496→      }
   497→    })();
   498→  }
