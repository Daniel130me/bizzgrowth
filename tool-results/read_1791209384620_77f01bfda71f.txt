   245→   245→function answerFor(question: string, type: BusinessType): AnswerContent {
   246→   246→  const m = MATCHERS[type];
   247→   247→  const hit = m.list.find((c) => c.test.test(question));
   248→   248→  return hit ? hit.content : m.fallback;
   249→   249→}
